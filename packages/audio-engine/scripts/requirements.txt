# Python runtime dependencies for Sound Buddy's audio-engine scripts.
#
# spectrum.py — offline frequency-band / spectral analysis
# stream.py   — live capture + real-time analysis
#
# Install into a virtualenv:
#   python3 -m venv .venv && source .venv/bin/activate
#   pip install -r packages/audio-engine/scripts/requirements.txt
#
# The Electron app invokes `python3` from PATH, so that interpreter must have
# these installed (activate the venv before launching, or point the app at it).
#
# numpy is the only heavy dependency: all DSP (spectrum.py, stream.py) is
# numpy-only, so the packaged runtime ships no scipy (#665).

numpy>=1.24
soundfile>=0.12      # spectrum.py: WAV/FLAC/AIFF/OGG decode (ffmpeg subprocess fallback for m4a/aac)
sounddevice>=0.4     # stream.py: live audio capture
